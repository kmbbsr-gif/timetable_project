def get_effective_school(request):
    if request.user.is_superuser:
        return getattr(request, "school", None)
    return get_user_school(request.user)


def verify_school_access(request, obj_school):
    if request.user.is_superuser:
        return True
    return obj_school == get_user_school(request.user)


def school_scoped(qs, request, field_name="school"):
    user_school = get_effective_school(request)
    if user_school:
        return qs.filter(**{field_name: user_school})
    if request.user.is_superuser:
        return qs
    return qs.none()


def is_admin_or_coordinator(user):
    """Helper function checking if user belongs to Admin or Coordinator groups or is a superuser."""
    return user.is_superuser or user.groups.filter(name__in=['Admin', 'Coordinator']).exists()


# ---------- Timetable Generator ----------
@login_required
@group_required('Admin', 'Coordinator')
def generate_timetable(request):
    if request.method == 'POST':
        school_id = request.POST.get('school_id')
        session_id = request.POST.get('session_id')
        allow_partial = request.POST.get('allow_partial') == 'true'

        if not school_id or not session_id:
            messages.error(request, "Please select a school and session.")
            return render(request, 'timetable/generate_form.html', {
                'schools': school_scoped(School.objects.all(), request),
                'sessions': AcademicSession.objects.filter(school__in=school_scoped(School.objects.all(), request)),
            })
        try:
            solver = TimetableSolver(int(school_id), int(session_id))
            
            # --- Diagnostic Check for Partial Combined Group Overlaps ---
            if hasattr(solver, 'analyze_combined_group_availability'):
                partial_reports = solver.analyze_combined_group_availability()

                if partial_reports and not allow_partial:
                    return render(request, 'timetable/generate_form.html', {
                        'schools': school_scoped(School.objects.all(), request),
                        'sessions': AcademicSession.objects.filter(school__in=school_scoped(School.objects.all(), request)),
                        'partial_reports': partial_reports,
                        'selected_school_id': school_id,
                        'selected_session_id': session_id,
                    })

            # --- Solve Timetable ---
            count = solver.solve(use_genetic=False)
            if count:
                messages.success(request, f"✅ Timetable generated! {count} entries created.")
                return redirect('dashboard')
            else:
                messages.error(request, "❌ No solution found. Check constraints.")
                return render(request, 'timetable/generate_form.html', {
                    'schools': school_scoped(School.objects.all(), request),
                    'sessions': AcademicSession.objects.filter(school__in=school_scoped(School.objects.all(), request))),
                })
        except Exception as e:
            messages.error(request, f"⚠️ Error: {str(e)}")
            traceback.print_exc()
            return render(request, 'timetable/generate_form.html', {
                'schools': school_scoped(School.objects.all(), request),
                'sessions': AcademicSession.objects.filter(school__in=school_scoped(School.objects.all(), request)),
            })

    schools = school_scoped(School.objects.all(), request)
    sessions = AcademicSession.objects.filter(school__in=schools)
    return render(request, 'timetable/generate_form.html', {'schools': schools, 'sessions': sessions})

@login_required
@group_required('Admin', 'Coordinator')
def timetable_index(request):
    user_school = get_effective_school(request)
    all_classes = school_scoped(Class.objects.filter(is_active=True), request).order_by('display_order')
    class_sections = []
    sections_data_json = []
    for cls in all_classes:
        sections = Section.objects.filter(class_instance=cls, is_active=True).order_by('name')
        class_sections.append({'class': cls, 'sections': sections})
        sections_data_json.append({
            'class': {'id': cls.id, 'name': cls.name},
            'sections': [{'id': s.id, 'name': s.name} for s in sections]
        })
        tt_qs = school_scoped(TimetableEntry.objects.all(), request)
    classes_with_entries = school_scoped(
        Class.objects.filter(id__in=tt_qs.values_list('class_instance_id', flat=True).distinct()),
        request
    ).order_by('display_order')
    
    data = []
    for cls in classes_with_entries:
        sections = Section.objects.filter(
            class_instance=cls,
            id__in=tt_qs.filter(class_instance=cls).values_list('section_id', flat=True).distinct()
        ).order_by('name')
        data.append({'class': cls, 'sections': sections})
    context = {
        'class_sections': class_sections,
        'sections_data_json': sections_data_json,
        'data': data,
    }
    return render(request, 'timetable/index.html', context)

# ---------- Class Timetable ----------
def view_class_timetable(request, class_id, section_id):
        user_school = get_effective_school(request)
    if user_school:
        class_obj = get_object_or_404(Class, id=class_id, school=user_school)
        section = get_object_or_404(Section, id=section_id, class_instance__school=user_school)
    else:
    date_str = request.GET.get('date')
    date = None
    if date_str:
        try:
            date = datetime.strptime(date_str, '%Y-%m-%d').date()
        except ValueError:
            pass

    class_obj = get_object_or_404(Class, id=class_id)
    section = get_object_or_404(Section, id=section_id)

    entries = TimetableEntry.objects.filter(
        class_instance=class_obj,
        section=section
    ).select_related('teacher', 'subject')

    all_days = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday']
    if date is not None:
        day_idx = date.weekday()
        if day_idx >= 6:
            day_idx = 0
        days_to_show = [(day_idx, all_days[day_idx])]
    else:
        days_to_show = [(i, name) for i, name in enumerate(all_days)]

    grid = {}
    for entry in entries:
        key = f"{entry.day_of_week}_{entry.period_number}"
        grid[key] = entry

    periods = PeriodDefinition.objects.filter(
        school=class_obj.school,
        wing=section.wing
    ).order_by('period_number')
    if not periods:
        periods = PeriodDefinition.objects.filter(school=class_obj.school).order_by('period_number')
    if not periods:
        periods = [{'period_number': i, 'start_time': None, 'end_time': None} for i in range(1, 11)]

    table = []
    for p in periods:
        pn = p.period_number if hasattr(p, 'period_number') else p['period_number']
        row = {'period': p, 'cells': []}
        for day_idx, _ in days_to_show:
            key = f"{day_idx}_{pn}"
            row['cells'].append(grid.get(key))
        table.append(row)

    context = {
        'class': class_obj,
        'section': section,
        'table': table,
        'days_with_indices': days_to_show,
        'selected_date': date_str,
    }
    return render(request, 'timetable/class_view.html', context)

@login_required
@group_required('Admin', 'Coordinator')
def master_timetable(request):
        user_school = get_effective_school(request)
    if user_school:
        school = user_school
    elif request.user.is_superuser:
        school = School.objects.first()
    else:
        return render(request, 'timetable/master.html', {'data': [], 'days_with_indices': []})
    if not school:
        return render(request, 'timetable/master.html', {'data': [], 'days_with_indices': []})

    classes = Class.objects.filter(school=school, is_active=True).order_by('display_order')
    period_numbers = PeriodDefinition.objects.filter(school=school).values_list('period_number', flat=True).distinct()
    if not period_numbers:
        period_numbers = range(1, 11)

    periods = []
    for pn in sorted(set(period_numbers)):
        p_def = PeriodDefinition.objects.filter(school=school, period_number=pn).first()
        if p_def:
            periods.append(p_def)
        else:
            periods.append({'period_number': pn, 'start_time': None, 'end_time': None})

    days = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday']
    days_with_indices = [(i, name) for i, name in enumerate(days)]

    data = []
    for cls in classes:
        sections = Section.objects.filter(class_instance=cls, is_active=True).order_by('name')
        for section in sections:
            entries = TimetableEntry.objects.filter(
                class_instance=cls,
                section=section
            ).select_related('subject', 'teacher')
            grid = {}
            for e in entries:
                key = f"{e.day_of_week}_{e.period_number}"
                grid[key] = e
            table = []
            for p in periods:
                pn = p.period_number if hasattr(p, 'period_number') else p['period_number']
                row = {'period': p, 'cells': []}
                for day_idx, _ in days_with_indices:
                    key = f"{day_idx}_{pn}"
                    row['cells'].append(grid.get(key))
                table.append(row)
            data.append({
                'class': cls,
                'section': section,
                'table': table,
            })

    teachers = Teacher.objects.filter(school=school, is_active=True).select_related('school')
    teacher_list = []
    for t in teachers:
        subjects = TeacherSubjectAssignment.objects.filter(teacher=t).values_list('class_subject__subject__name', flat=True).distinct()
        teacher_list.append({
            'id': t.id,
            'name': t.name,
            'subjects': list(subjects),
        })

    total_required = 0
        for cs in ClassSubject.objects.filter(class_instance__school=school, class_instance__is_active=True):
        section_count = Section.objects.filter(class_instance=cs.class_instance, is_active=True).count()
        total_required += cs.weekly_periods * section_count

        total_assigned = TeacherSubjectAssignment.objects.filter(
        class_subject__class_instance__school=school
    ).aggregate(total=Sum('weekly_periods'))['total'] or 0

    teaching_slots = 0
    total_slots = 0
    if school:
            sections = Section.objects.filter(class_instance__school=school, is_active=True)
        for section in sections:
            periods_def = PeriodDefinition.objects.filter(school=school, wing=section.wing)
            periods_per_day = periods_def.count()
            teaching_periods = periods_def.filter(is_assembly=False, is_lunch=False, is_break=False)
            teaching_per_day = teaching_periods.count()
            if periods_per_day > 0:
                working_days = section.working_days or list(range(0, 6))
                total_slots += periods_per_day * len(working_days)
                teaching_slots += teaching_per_day * len(working_days)

        active_teacher_count = Teacher.objects.filter(school=school, is_active=True).count()
    average_per_teacher = round(teaching_slots / active_teacher_count, 1) if active_teacher_count > 0 else 0

    context = {
        'data': data,
        'days_with_indices': days_with_indices,
        'total_required': total_required,
        'total_assigned': total_assigned,
        'total_remaining': total_required - total_assigned,
        'teaching_slots': teaching_slots,
        'total_slots': total_slots,
        'active_teacher_count': active_teacher_count,
        'average_per_teacher': average_per_teacher,
        'teacher_list': teacher_list,
    }
    return render(request, 'timetable/master.html', context)


# ---------- Free Periods ----------
def free_periods(request):
        user_school = get_effective_school(request)
    if user_school:
        school = user_school
    elif request.user.is_superuser:
        school = School.objects.first()
    else:
        return render(request, 'timetable/free_periods.html', {'days_with_indices': [], 'table': [], 'teachers': []})
    if not school:
        return render(request, 'timetable/free_periods.html', {'days_with_indices': [], 'table': [], 'teachers': []})

    all_period_numbers = PeriodDefinition.objects.filter(school=school).values_list('period_number', flat=True).distinct()
    if not all_period_numbers:
        all_period_numbers = range(1, 11)

    periods = []
    for pn in sorted(set(all_period_numbers)):
        p_def = PeriodDefinition.objects.filter(school=school, period_number=pn).first()
        if p_def:
            periods.append(p_def)
        else:
            periods.append({'period_number': pn, 'start_time': None, 'end_time': None})

    days = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday']
    days_with_indices = [(i, name) for i, name in enumerate(days)]
    teachers = Teacher.objects.filter(is_active=True, school=school)

    occupied = set()
        for t_id, day, period in TimetableEntry.objects.filter(school=school).values_list('teacher_id', 'day_of_week', 'period_number'):
        occupied.add((t_id, day, period))

    table = []
    for p in periods:
        pn = p.period_number if hasattr(p, 'period_number') else p['period_number']
        row = {'period': p, 'cells': []}
        for day_idx, _ in days_with_indices:
            free_list = []
            for t in teachers:
                if (t.id, day_idx, pn) not in occupied:
                    free_list.append(t.name)
            row['cells'].append(free_list)
        table.append(row)

    selected_teacher_id = request.GET.get('teacher_id')
    selected_day = request.GET.get('day')
    teacher_free_info = None
    selected_teacher = None

    if selected_teacher_id:
        try:
            selected_teacher = Teacher.objects.get(id=int(selected_teacher_id), school=school)
            occupied_teacher = set()
            for entry in TimetableEntry.objects.filter(teacher=selected_teacher):
                occupied_teacher.add((entry.day_of_week, entry.period_number))

            if selected_day == 'all' or selected_day is None:
                free_by_day = {}
                for day_idx, day_name in days_with_indices:
                    free_periods_list = []
                    for p in periods:
                        pn = p.period_number if hasattr(p, 'period_number') else p['period_number']
                        if (day_idx, pn) not in occupied_teacher:
                            free_periods_list.append(pn)
                    free_by_day[day_name] = free_periods_list
                teacher_free_info = {'type': 'all_days', 'data': free_by_day}
            else:
                day_idx = int(selected_day)
                day_name = days[day_idx]
                free_periods_list = []
                for p in periods:
                    pn = p.period_number if hasattr(p, 'period_number') else p['period_number']
                    if (day_idx, pn) not in occupied_teacher:
                        free_periods_list.append(pn)
                teacher_free_info = {'type': 'single_day', 'day': day_name, 'periods': free_periods_list}
        except (ValueError, Teacher.DoesNotExist):
            pass

    context = {
        'days_with_indices': days_with_indices,
        'table': table,
        'teachers': teachers,
        'selected_teacher': selected_teacher,
        'selected_day': selected_day,
        'teacher_free_info': teacher_free_info,
        'periods': periods,
    }
    return render(request, 'timetable/free_periods.html', context)

# ---------- Teacher Absence Management ----------
@login_required
@group_required('Admin', 'Coordinator')
def manage_absence(request):
    user_school = get_effective_school(request)
    teacher_id = request.GET.get('teacher_id')
    date_str = request.GET.get('date')
    selected_date = None
    day_of_week = None
    teacher = None
    schedule = []
    free_teachers_by_period = {}
    substitutions = {}

    if teacher_id and date_str:
        try:
            teacher = get_object_or_404(Teacher, id=int(teacher_id))
            selected_date = datetime.strptime(date_str, '%Y-%m-%d').date()
            day_of_week = selected_date.weekday()

            period_numbers = TimetableEntry.objects.filter(
                teacher=teacher,
                day_of_week=day_of_week
            ).values_list('period_number', flat=True).distinct()
            
            periods = PeriodDefinition.objects.filter(
                school=teacher.school,
                period_number__in=period_numbers
            ).order_by('period_number')

            entries = TimetableEntry.objects.filter(
                teacher=teacher,
                day_of_week=day_of_week
            ).select_related('class_instance', 'section', 'subject')

            for entry in entries:
                schedule.append({
                    'period': entry.period_number,
                    'class_instance': entry.class_instance,
                    'section': entry.section,
                    'subject': entry.subject,
                    'entry': entry,
                })

            all_teachers = Teacher.objects.filter(is_active=True, school=teacher.school).exclude(id=teacher.id)
            occupied = {t.id: set() for t in all_teachers}
            
            for t in all_teachers:
                occupied_periods = TimetableEntry.objects.filter(
                    teacher=t,
                    day_of_week=day_of_week
                ).values_list('period_number', flat=True)
                occupied[t.id] = set(occupied_periods)

            free_teachers_by_period = {}
            for entry in schedule:
                pn = entry['period']
                free_list = [t for t in all_teachers if pn not in occupied[t.id]]
                free_teachers_by_period[pn] = free_list
                entry['free_teachers'] = free_list

            sub_assignments = SubstituteAssignment.objects.filter(
                absent_teacher=teacher,
                date=selected_date
            ).select_related('substitute_teacher', 'class_instance', 'section', 'subject')
            
            for sub in sub_assignments:
                substitutions[(sub.period_number, sub.class_instance.id, sub.section.id)] = sub.substitute_teacher

        except ValueError:
            messages.error(request, "Invalid data.")
            return redirect('timetable_index')

    if request.method == 'POST':
        teacher_id = request.POST.get('teacher_id')
        date_str = request.POST.get('date')
        if not teacher_id or not date_str:
            messages.error(request, "Missing teacher or date.")
            return redirect('manage_absence')
        try:
            teacher = get_object_or_404(Teacher, id=int(teacher_id))
            selected_date = datetime.strptime(date_str, '%Y-%m-%d').date()
            for key, value in request.POST.items():
                if key.startswith('sub_'):
                    parts = key.split('_')
                    if len(parts) == 4:
                        period = int(parts[1])
                        class_id = int(parts[2])
                        section_id = int(parts[3])
                        substitute_id = int(value) if value else None

                        orig_entry = TimetableEntry.objects.filter(
                            teacher=teacher,
                            day_of_week=selected_date.weekday(),
                            period_number=period,
                            class_instance_id=class_id,
                            section_id=section_id
                        ).first()
                        if not orig_entry:
                            continue

                        SubstituteAssignment.objects.filter(
                            absent_teacher=teacher,
                            date=selected_date,
                            period_number=period,
                            class_instance_id=class_id,
                            section_id=section_id
                        ).delete()

                        if substitute_id:
                            substitute_teacher = get_object_or_404(Teacher, id=substitute_id)
                            SubstituteAssignment.objects.create(
                                absent_teacher=teacher,
                                substitute_teacher=substitute_teacher,
                                date=selected_date,
                                period_number=period,
                                class_instance=orig_entry.class_instance,
                                section=orig_entry.section,
                                subject=orig_entry.subject,
                            )
            messages.success(request, "Substitutions saved successfully.")
            return redirect(f"{request.path}?teacher_id={teacher_id}&date={date_str}")
        except Exception as e:
            messages.error(request, f"Error: {str(e)}")

    teachers = Teacher.objects.filter(is_active=True).order_by('name')
    context = {
        'teachers': teachers,
        'selected_teacher': teacher,
        'selected_date': selected_date,
        'schedule': schedule,
        'free_teachers_by_period': free_teachers_by_period,
        'substitutions': substitutions,
        'periods': [entry['period'] for entry in schedule],
    }
    return render(request, 'timetable/manage_absence.html', context)

def get_conflict_suggestions(source_entry, target_entry=None, target_day=None, target_period=None):
    school = source_entry.school
    if target_entry:
        target_day = target_entry.day_of_week
        target_period = target_entry.period_number
    else:
        target_day = int(target_day) if target_day is not None else None
        target_period = int(target_period) if target_period is not None else None

    suggestions = {
        'free_teachers': [],
        'free_periods': [],
    }

    if target_day is not None and target_period is not None:
        all_teachers = Teacher.objects.filter(school=school, is_active=True).exclude(id=source_entry.teacher_id)
        occupied_teachers = TimetableEntry.objects.filter(
            school=school,
            day_of_week=target_day,
            period_number=target_period
        ).values_list('teacher_id', flat=True)
        free_teachers = all_teachers.exclude(id__in=occupied_teachers)
        suggestions['free_teachers'] = list(free_teachers.values('id', 'name')[:10])

        source_class = source_entry.class_instance
        source_section = source_entry.section
        occupied_slots = TimetableEntry.objects.filter(
            class_instance=source_class,
            section=source_section
        ).values_list('day_of_week', 'period_number')
        occupied_set = set(occupied_slots)
        periods = PeriodDefinition.objects.filter(school=school).values_list('period_number', flat=True).distinct()
        days = range(6)
        free_slots = []
        for day in days:
            for pn in periods:
                if (day, pn) not in occupied_set:
                    free_slots.append({'day': day, 'period': pn})
        suggestions['free_periods'] = free_slots[:10]

    return suggestions


def get_temp_slot(school, session, excluded_ids):
    occupied = set(
        TimetableEntry.objects.filter(school=school, session=session)
        .exclude(id__in=excluded_ids)
        .values_list('day_of_week', 'period_number')
    )
    for day in [6, 5, 4, 3, 2, 1, 0]:
        for period in [99, 98, 97, 96]:
            if (day, period) not in occupied:
                return day, period
    for day in range(7):
        for period in range(1, 21):
            if (day, period) not in occupied:
                return day, period
    raise Exception("No free temporary slot found")

@login_required
@group_required('Admin', 'Coordinator')
@csrf_exempt
def swap_entries(request):
    if request.method != 'POST':
        return JsonResponse({'error': 'Method not allowed'}, status=405)

    source_id = request.POST.get('source_id')
    target_id = request.POST.get('target_id')
    move = request.POST.get('move') == 'true'
    override = request.POST.get('override') == 'true'
    swap_teachers = request.POST.get('swap_teachers') == 'true'
    assign_free_teacher = request.POST.get('assign_free_teacher')

    if not source_id:
        return JsonResponse({'error': 'Missing source entry'}, status=400)

    try:
        source = TimetableEntry.objects.get(id=source_id)
        if not verify_school_access(request, source.school):
        return JsonResponse({'error': 'Access denied'}, status=403)
    except TimetableEntry.DoesNotExist:
        return JsonResponse({'error': 'Source entry not found'}, status=404)

    conflicts = []

    def success_response():
        version, _ = TimetableVersion.objects.get_or_create(
            school=source.school,
            session=source.session
        )
        version.save()
        return JsonResponse({'success': True})

    if assign_free_teacher:
        try:
            new_teacher = Teacher.objects.get(id=int(assign_free_teacher))
                        if not verify_school_access(request, new_teacher.school):
                return JsonResponse({'error': 'Access denied'}, status=403)
            source.teacher = new_teacher
            source.save()
            return success_response()
        except Teacher.DoesNotExist:
            return JsonResponse({'error': 'Teacher not found'}, status=404)

    if swap_teachers and target_id:
        try:
            target = TimetableEntry.objects.get(id=target_id)
                    if not verify_school_access(request, target.school):
            return JsonResponse({'error': 'Access denied'}, status=403)
        except TimetableEntry.DoesNotExist:
            return JsonResponse({'error': 'Target entry not found'}, status=404)
        source_teacher = source.teacher
        target_teacher = target.teacher
        source.teacher = target_teacher
        target.teacher = source_teacher
        with transaction.atomic():
            source.save()
            target.save()
        return success_response()

    if move:
        target_class_id = request.POST.get('target_class')
        target_section_id = request.POST.get('target_section')
        target_day = request.POST.get('target_day')
        target_period = request.POST.get('target_period')

        if not all([target_class_id, target_section_id, target_day, target_period]):
            return JsonResponse({'error': 'Missing target position'}, status=400)

        existing = TimetableEntry.objects.filter(
            class_instance_id=target_class_id,
            section_id=target_section_id,
            day_of_week=target_day,
            period_number=target_period
        ).exists()
        if existing:
            return JsonResponse({'error': 'Target cell is not empty'}, status=400)

        overlap_teacher = TimetableEntry.objects.filter(
            teacher=source.teacher,
            day_of_week=target_day,
            period_number=target_period
        ).exclude(id=source.id)
        
        if overlap_teacher.exists() and not source.is_combined:
            conflicts.append(f"Teacher {source.teacher.name} already has a class at that time")

        if conflicts and not override:
            suggestions = get_conflict_suggestions(source, target_day=int(target_day), target_period=int(target_period))
            return JsonResponse({
                'conflicts': conflicts,
                'requires_override': True,
                'suggestions': suggestions,
            }, status=409)

        with transaction.atomic():
            source.day_of_week = int(target_day)
            source.period_number = int(target_period)
            source.save()
        return success_response()

    if not move and target_id:
        try:
            target = TimetableEntry.objects.get(id=target_id)
                    if not verify_school_access(request, target.school):
            return JsonResponse({'error': 'Access denied'}, status=403)
        except TimetableEntry.DoesNotExist:
            return JsonResponse({'error': 'Target entry not found'}, status=404)

        overlap_teacher_src = TimetableEntry.objects.filter(
            teacher=source.teacher,
            day_of_week=target.day_of_week,
            period_number=target.period_number
        ).exclude(id=source.id).exclude(id=target.id)
        if overlap_teacher_src.exists() and not source.is_combined:
            conflicts.append(f"Teacher {source.teacher.name} already has a class at {target.get_day_of_week_display()} P{target.period_number}")

        overlap_teacher_tgt = TimetableEntry.objects.filter(
            teacher=target.teacher,
            day_of_week=source.day_of_week,
            period_number=source.period_number
        ).exclude(id=source.id).exclude(id=target.id)
        if overlap_teacher_tgt.exists() and not target.is_combined:
            conflicts.append(f"Teacher {target.teacher.name} already has a class at {source.get_day_of_week_display()} P{source.period_number}")

        overlap_class_src = TimetableEntry.objects.filter(
            class_instance=source.class_instance,
            section=source.section,
            day_of_week=target.day_of_week,
            period_number=target.period_number
        ).exclude(id=source.id).exclude(id=target.id)
        if overlap_class_src.exists():
            conflicts.append(f"Class {source.class_instance.name} {source.section.name} already has a class at {target.get_day_of_week_display()} P{target.period_number}")

        overlap_class_tgt = TimetableEntry.objects.filter(
            class_instance=target.class_instance,
            section=target.section,
            day_of_week=source.day_of_week,
            period_number=source.period_number
        ).exclude(id=source.id).exclude(id=target.id)
        if overlap_class_tgt.exists():
            conflicts.append(f"Class {target.class_instance.name} {target.section.name} already has a class at {source.get_day_of_week_display()} P{source.period_number}")

        if source.room:
            overlap_room_src = TimetableEntry.objects.filter(
                room=source.room,
                day_of_week=target.day_of_week,
                period_number=target.period_number
            ).exclude(id=source.id).exclude(id=target.id)
            if overlap_room_src.exists():
                conflicts.append(f"Room {source.room.name} already occupied at {target.get_day_of_week_display()} P{target.period_number}")

        if target.room:
            overlap_room_tgt = TimetableEntry.objects.filter(
                room=target.room,
                day_of_week=source.day_of_week,
                period_number=source.period_number
            ).exclude(id=source.id).exclude(id=target.id)
            if overlap_room_tgt.exists():
                conflicts.append(f"Room {target.room.name} already occupied at {source.get_day_of_week_display()} P{source.period_number}")

        if conflicts and not override:
            suggestions = get_conflict_suggestions(source, target)
            return JsonResponse({
                'conflicts': conflicts,
                'requires_override': True,
                'suggestions': suggestions,
            }, status=409)

        with transaction.atomic():
            src_day, src_period = source.day_of_week, source.period_number
            tgt_day, tgt_period = target.day_of_week, target.period_number

            temp_day, temp_period = get_temp_slot(source.school, source.session, [source.id, target.id])

            source.day_of_week = temp_day
            source.period_number = temp_period
            source.save()

            target.day_of_week = src_day
            target.period_number = src_period
            target.save()

            source.day_of_week = tgt_day
            source.period_number = tgt_period
            source.save()

        return success_response()

    return JsonResponse({'error': 'Invalid operation'}, status=400)

@login_required
@user_passes_test(is_admin_or_coordinator)
@csrf_exempt
@require_POST
def assign_teacher_to_slot(request):
    user_school = get_effective_school(request)
    teacher_id = request.POST.get('teacher_id')
    class_id = request.POST.get('class_id')
    section_id = request.POST.get('section_id')
    day = request.POST.get('day')
    period = request.POST.get('period')
    subject_id = request.POST.get('subject_id')
    new_subject_name = (request.POST.get('new_subject_name') or '').strip()
    is_combined = request.POST.get('is_combined', 'false').lower() == 'true'

    if not all([teacher_id, class_id, section_id, day, period]):
        return JsonResponse({'success': False, 'error': 'Missing required parameters.'}, status=400)

    try:
        teacher = Teacher.objects.get(id=teacher_id)
        class_obj = Class.objects.get(id=class_id)
        section = Section.objects.get(id=section_id)
        school = class_obj.school

        session = AcademicSession.objects.filter(school=school, is_current=True).first()
        if not session:
            return JsonResponse({'success': False, 'error': 'No active academic session found.'}, status=400)

        # Teacher conflict check
        teacher_conflict = TimetableEntry.objects.filter(
            teacher=teacher,
            day_of_week=day,
            period_number=period
        ).first()

        if teacher_conflict and not (is_combined or teacher_conflict.is_combined):
            t_name = teacher_conflict.teacher.name
            c_name = teacher_conflict.class_instance.name
            s_name = teacher_conflict.subject.name if teacher_conflict.subject else 'Subject'
            return JsonResponse({
                'success': False,
                'error': f"{t_name} is already teaching {s_name} in {c_name} during Period {period} on this day! Enable 'Combined Class Mode' in the deck to link sections."
            }, status=409)

        if subject_id or new_subject_name:
            with transaction.atomic():
                if new_subject_name:
                    subject = Subject.objects.filter(school=school, name__iexact=new_subject_name).first()
                    if not subject:
                        subject = Subject.objects.create(
                            school=school,
                            name=new_subject_name,
                            code=new_subject_name[:10].upper()
                        )
                    class_subject, _ = ClassSubject.objects.get_or_create(
                        class_instance=class_obj,
                        subject=subject,
                        defaults={
                            'weekly_periods': 1,
                            'double_period_allowed': False,
                            'practical_required': False,
                            'lab_required': False
                        }
                    )
                else:
                    subject = Subject.objects.get(id=subject_id, school=school)
                    class_subject = ClassSubject.objects.filter(class_instance=class_obj, subject=subject).first()
                    if not class_subject:
                        return JsonResponse({'success': False, 'error': 'Subject is not part of this class curriculum.'}, status=400)

                TeacherSubjectAssignment.objects.get_or_create(
                    teacher=teacher,
                    class_subject=class_subject,
                    defaults={
                        'weekly_periods': 1,
                        'is_primary': False,
                        'preference_weight': 1.0
                    }
                )

                TimetableEntry.objects.update_or_create(
                    school=school,
                    session=session,
                    class_instance=class_obj,
                    section=section,
                    day_of_week=day,
                    period_number=period,
                    defaults={
                        'teacher': teacher,
                        'subject': subject,
                        'is_combined': False,
                        'combined_group_id': None
                    }
                )

                version, _ = TimetableVersion.objects.get_or_create(school=school, session=session)
                version.save()

            return JsonResponse({'success': True, 'message': 'Assignment created successfully.'})

        valid_assignments = TeacherSubjectAssignment.objects.filter(
            teacher=teacher,
            class_subject__class_instance=class_obj
        ).select_related('class_subject__subject')

        if valid_assignments.count() == 1:
            subject = valid_assignments.first().class_subject.subject
        elif valid_assignments.count() > 1:
            seen_ids = set()
            subjects_list = []
            for va in valid_assignments:
                sid = va.class_subject.subject.id
                if sid not in seen_ids:
                    seen_ids.add(sid)
                    subjects_list.append({
                        'id': sid,
                        'name': va.class_subject.subject.name,
                        'code': va.class_subject.subject.code or va.class_subject.subject.name[:10]
                    })
            return JsonResponse({
                'success': False,
                'requires_selection': True,
                'mode': 'select_existing',
                'subjects': subjects_list,
                'allow_new': False,
                'message': 'This teacher teaches multiple subjects for this class. Choose one:'
            })
        else:
            class_subjects = ClassSubject.objects.filter(
                class_instance=class_obj
            ).select_related('subject')

            subjects_list = []
            for cs in class_subjects:
                subjects_list.append({
                    'id': cs.subject.id,
                    'name': cs.subject.name,
                    'code': cs.subject.code or cs.subject.name[:10]
                })

            return JsonResponse({
                'success': False,
                'requires_selection': True,
                'mode': 'assign_new',
                'subjects': subjects_list,
                'allow_new': True,
                'message': 'Teacher has no subjects assigned for this class. Select an existing subject or create a new one.'
            })

    except Exception as e:
        traceback.print_exc()
        return JsonResponse({'success': False, 'error': str(e)}, status=500)

@login_required
@group_required('Admin', 'Coordinator')
def create_fixed_entries(request):
    school = getattr(request, 'school', None)
    if not school and request.user.is_superuser:
        school = School.objects.first()

    if not school:
        messages.error(request, "No school found for the active workspace. Please select or set up a school.")
        return redirect('dashboard')

    session = AcademicSession.objects.filter(school=school, is_current=True).first()
    if not session:
        messages.error(request, "No active academic session found for this school. Please set one up first.")
        return redirect('dashboard')

    if request.method == 'POST':
        form = FixedEntryForm(request.POST, school=school, user=request.user)
        if form.is_valid():
            data = form.cleaned_data
            apply_all = data['apply_to_all']
            section = data['section']
            day = int(data['day'])
            period = data['period']
            activity = data['activity_name'].strip()

            subject, _ = Subject.objects.get_or_create(
                school=school,
                name=activity,
                defaults={'code': activity[:10].upper()}
            )

            unique_emp_id = f"FIX_{activity[:6].upper()}_{uuid.uuid4().hex[:4].upper()}"
            teacher, _ = Teacher.objects.get_or_create(
                school=school,
                name=f"Fixed: {activity}",
                defaults={'employee_id': unique_emp_id}
            )

            if apply_all:
                target_sections = Section.objects.filter(class_instance__school=school, is_active=True).select_related('class_instance')
            else:
                target_sections = [section]

            created_count = 0
            with transaction.atomic():
                for sec in target_sections:
                    cls_obj = sec.class_instance
                    existing = TimetableEntry.objects.filter(
                        school=school,
                        session=session,
                        class_instance=cls_obj,
                        section=sec,
                        day_of_week=day,
                        period_number=period,
                        is_fixed=True
                    ).exists()
                    if not existing:
                        TimetableEntry.objects.create(
                            school=school,
                            session=session,
                            class_instance=cls_obj,
                            section=sec,
                            day_of_week=day,
                            period_number=period,
                            teacher=teacher,
                            subject=subject,
                            is_fixed=True,
                            is_double=False,
                            is_practical=False
                        )
                        created_count += 1

                version, _ = TimetableVersion.objects.get_or_create(school=school, session=session)
                version.save()

            day_dict = dict(DAY_CHOICES)
            day_name = day_dict.get(day, day_dict.get(str(day), f"Day {day}"))
            messages.success(request, f"Created {created_count} fixed entries for '{activity}' on {day_name} P{period}.")
            return redirect('manage_fixed_entries')
    else:
        form = FixedEntryForm(school=school, user=request.user)

    fixed_entries = TimetableEntry.objects.filter(
        school=school,
        is_fixed=True
    ).select_related(
        'class_instance', 'section', 'subject', 'teacher', 'school'
    ).order_by(
        'class_instance__display_order', 'section__name', 'day_of_week', 'period_number'
    )

    context = {
        'form': form,
        'fixed_entries': fixed_entries,
        'current_school': school,
    }
    return render(request, 'timetable/fixed_entries.html', context)


@login_required
@group_required('Admin', 'Coordinator')
def edit_fixed_entry(request, entry_id):
    school = getattr(request, 'school', None)
    if request.user.is_superuser:
        entry = get_object_or_404(TimetableEntry, id=entry_id, is_fixed=True)
    else:
        entry = get_object_or_404(TimetableEntry, id=entry_id, school=school, is_fixed=True)

    if request.method == 'POST':
        form = FixedEntryEditForm(request.POST)
        if form.is_valid():
            data = form.cleaned_data
            new_day = int(data['day'])
            new_period = data['period']
            new_activity = data['activity_name'].strip()

            existing = TimetableEntry.objects.filter(
                school=entry.school,
                session=entry.session,
                class_instance=entry.class_instance,
                section=entry.section,
                day_of_week=new_day,
                period_number=new_period
            ).exclude(id=entry.id).exists()

            if existing:
                messages.error(request, "This slot is already occupied. Choose another day or period.")
                return render(request, 'timetable/fixed_edit.html', {'form': form, 'entry': entry})

            entry.day_of_week = new_day
            entry.period_number = new_period

            if new_activity != entry.subject.name:
                subject, _ = Subject.objects.get_or_create(
                    school=entry.school,
                    name=new_activity,
                    defaults={'code': new_activity[:10].upper()}
                )
                unique_emp_id = f"FIX_{new_activity[:6].upper()}_{uuid.uuid4().hex[:4].upper()}"
                teacher, _ = Teacher.objects.get_or_create(
                    school=entry.school,
                    name=f"Fixed: {new_activity}",
                    defaults={'employee_id': unique_emp_id}
                )
                entry.subject = subject
                entry.teacher = teacher

            entry.save()
            version, _ = TimetableVersion.objects.get_or_create(school=entry.school, session=entry.session)
            version.save()
            messages.success(request, "Fixed entry updated successfully.")
            return redirect('manage_fixed_entries')
    else:
        initial_data = {
            'day': entry.day_of_week,
            'period': entry.period_number,
            'activity_name': entry.subject.name,
        }
        form = FixedEntryEditForm(initial=initial_data)

    context = {
        'form': form,
        'entry': entry,

@login_required
@group_required('Admin', 'Coordinator', 'Teacher')
def class_summary(request, class_id=None, section_id=None):
    classes = Class.objects.filter(is_active=True).order_by('display_order')
    selected_class = None
    selected_section = None
    grid = {}
    summary = {}
    periods = []
    days = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday']
    days_with_indices = [(i, name) for i, name in enumerate(days)]

    if class_id is None:
        class_id = request.GET.get('class_id')
    if section_id is None:
        section_id = request.GET.get('section_id')

    sections_for_class = []
    if class_id:
        selected_class = get_object_or_404(Class, id=class_id)
        sections_for_class = Section.objects.filter(class_instance=selected_class, is_active=True).order_by('name')

    if class_id and section_id:
        selected_section = get_object_or_404(Section, id=section_id)
        periods = PeriodDefinition.objects.filter(
            school=selected_class.school,
            wing=selected_section.wing
        ).order_by('period_number')

        entries = TimetableEntry.objects.filter(
            class_instance=selected_class,
            section=selected_section
        ).select_related('subject', 'teacher')

        grid = {f"{entry.day_of_week}_{entry.period_number}": entry for entry in entries}

        assigned_qs = TeacherSubjectAssignment.objects.filter(
            class_subject__class_instance=selected_class,
            class_subject__class_instance__sections=selected_section
        )
        assigned_per_subject = {}
        total_assigned = 0
        for ass in assigned_qs:
            subject_name = ass.class_subject.subject.name
            assigned_per_subject[subject_name] = assigned_per_subject.get(subject_name, 0) + ass.weekly_periods
            total_assigned += ass.weekly_periods

        total_teaching = 0
        non_teaching = 0
        scheduled_per_subject = {}
        free_slots = 0

        for p in periods:
            pn = p.period_number
            is_teaching = not (p.is_assembly or p.is_lunch or p.is_break)
            if is_teaching:
                total_teaching += 1
                for day_idx, _ in days_with_indices:
                    key = f"{day_idx}_{pn}"
                    entry = grid.get(key)
                    if entry:
                        subject_name = entry.subject.name
                        scheduled_per_subject[subject_name] = scheduled_per_subject.get(subject_name, 0) + 1
                    else:
                        free_slots += 1
            else:
                non_teaching += 1

        summary = {
            'total_teaching': total_teaching,
            'non_teaching': non_teaching,
            'assigned_per_subject': assigned_per_subject,
            'scheduled_per_subject': scheduled_per_subject,
            'total_assigned': total_assigned,
            'scheduled_count': len(entries),
            'free_slots': free_slots,
            'mismatch': total_assigned - len(entries)
        }

    context = {
        'classes': classes,
        'selected_class': selected_class,
        'selected_section': selected_section,
        'sections_for_class': sections_for_class,
        'grid': grid,
        'summary': summary,
        'days_with_indices': days_with_indices,
        'periods': periods if selected_class else [],
    }
    return render(request, 'timetable/class_summary.html', context)


@login_required
@group_required('Admin', 'Coordinator', 'Teacher')
def class_summary_list(request):
    classes = Class.objects.filter(is_active=True).order_by('display_order')
    data = []

    for cls in classes:
        sections = Section.objects.filter(class_instance=cls, is_active=True).order_by('name')
        for section in sections:
            wing = section.wing
            school = cls.school
            working_days = section.working_days or list(range(0, 6))  # 6 days (Mon-Sat)

            # Get period definitions for this wing
            periods = PeriodDefinition.objects.filter(school=school, wing=wing).order_by('period_number')
            if not periods:
                periods = PeriodDefinition.objects.filter(school=school).order_by('period_number')

            fixed_entries = TimetableEntry.objects.filter(
                class_instance=cls,
                section=section,
                is_fixed=True
            ).values_list('day_of_week', 'period_number')
            fixed_set = set(fixed_entries)

            total_teaching_weekly = 0
            non_teaching_weekly = 0

            # 💡 FIXED LOOP: Count non-teaching periods FIRST before checking fixed status
            for day in working_days:
                for p in periods:
                    pn = p.period_number
                    
                    # 1. If it's defined as Assembly, Lunch, or Break, it's ALWAYS non-teaching
                    if p.is_assembly or p.is_lunch or p.is_break:
                        non_teaching_weekly += 1
                    # 2. If it's a fixed activity (like MPT or Yoga), it's not a standard subject teaching slot
                    elif (day, pn) in fixed_set:
                        non_teaching_weekly += 1
                    # 3. Otherwise, it is an available teaching slot
                    else:
                        total_teaching_weekly += 1

            assigned_qs = TeacherSubjectAssignment.objects.filter(
                class_subject__class_instance=cls,
                class_subject__class_instance__sections=section
            )
            total_assigned = assigned_qs.aggregate(total=Sum('weekly_periods'))['total'] or 0

            entries = TimetableEntry.objects.filter(
                class_instance=cls,
                section=section
            ).exclude(is_fixed=True)
            scheduled_count = entries.count()

            teaching_period_numbers = set(
                p.period_number for p in periods if not (p.is_assembly or p.is_lunch or p.is_break)
            )

            occupied = set((e.day_of_week, e.period_number) for e in entries if e.period_number in teaching_period_numbers)

            free_slots = 0
            for day in working_days:
                for p in periods:
                    pn = p.period_number
                    if pn in teaching_period_numbers and (day, pn) not in occupied and (day, pn) not in fixed_set:
                        free_slots += 1

            data.append({
                'class': cls,
                'section': section,
                'total_teaching_weekly': total_teaching_weekly,
                'non_teaching_weekly': non_teaching_weekly,
                'total_assigned': total_assigned,
                'scheduled_count': scheduled_count,
                'free_slots_weekly': free_slots,
                'mismatch': total_assigned - scheduled_count,
            })

    return render(request, 'timetable/class_summary_list.html', {'data': data})

@login_required
@group_required('Admin', 'Coordinator', 'Teacher')
def class_summary_detail(request, class_id, section_id):
    cls = get_object_or_404(Class, id=class_id)
    section = get_object_or_404(Section, id=section_id)

    assignments = TeacherSubjectAssignment.objects.filter(
        class_subject__class_instance=cls,
        class_subject__class_instance__sections=section
    ).select_related('class_subject__subject', 'teacher')

    assigned_per_subject = {}
    total_assigned = 0
    for a in assignments:
        subject_name = a.class_subject.subject.name
        assigned_per_subject[subject_name] = assigned_per_subject.get(subject_name, 0) + a.weekly_periods
        total_assigned += a.weekly_periods

    entries = TimetableEntry.objects.filter(
        class_instance=cls,
        section=section
    ).exclude(is_fixed=True).select_related('subject')

    scheduled_per_subject = {}
    total_scheduled = 0
    for e in entries:
        subject_name = e.subject.name
        scheduled_per_subject[subject_name] = scheduled_per_subject.get(subject_name, 0) + 1
        total_scheduled += 1

    subject_data = []
    all_subjects = set(assigned_per_subject.keys()) | set(scheduled_per_subject.keys())
    for subject in sorted(all_subjects):
        assigned = assigned_per_subject.get(subject, 0)
        scheduled = scheduled_per_subject.get(subject, 0)
        subject_data.append({
            'name': subject,
            'assigned': assigned,
            'scheduled': scheduled,
            'difference': assigned - scheduled,
        })

    context = {
        'class': cls,
        'section': section,
        'subject_data': subject_data,
        'total_assigned': total_assigned,
        'total_scheduled': total_scheduled,
        'total_difference': total_assigned - total_scheduled,
    }
    return render(request, 'timetable/class_summary_detail.html', context)


# ---------- Free Slots & Date Filtering ----------
def free_slots(request):
    school = School.objects.first()
    if not school:
        return render(request, 'timetable/free_slots.html', {'data': []})

    classes = Class.objects.filter(school=school, is_active=True).order_by('display_order')
    days = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday']
    days_with_indices = [(i, name) for i, name in enumerate(days)]

    wing_periods = {}
    for wing in ['Junior', 'Middle', 'Senior']:
        periods = PeriodDefinition.objects.filter(school=school, wing=wing).order_by('period_number')
        if periods:
            wing_periods[wing] = periods
        else:
            periods = PeriodDefinition.objects.filter(school=school).order_by('period_number')
            if periods:
                wing_periods[wing] = periods

    data = []
    for cls in classes:
        sections = Section.objects.filter(class_instance=cls, is_active=True).order_by('name')
        for section in sections:
            wing = section.wing
            periods = wing_periods.get(wing, [])
            if not periods:
                continue

            entries = TimetableEntry.objects.filter(class_instance=cls, section=section)
            assigned = {(e.day_of_week, e.period_number): e for e in entries}

            table = []
            for p in periods:
                pn = p.period_number if hasattr(p, 'period_number') else p['period_number']
                row = {'period': p, 'cells': [assigned.get((day_idx, pn)) for day_idx, _ in days_with_indices]}
                table.append(row)

            data.append({
                'class': cls,
                'section': section,
                'table': table,
                'days_with_indices': days_with_indices,
            })

    total_required = sum(cs.weekly_periods * Section.objects.filter(class_instance=cs.class_instance, is_active=True).count() for cs in ClassSubject.objects.filter(class_instance__is_active=True))
    total_assigned = TeacherSubjectAssignment.objects.aggregate(total=Sum('weekly_periods'))['total'] or 0

    teaching_slots = 0
    total_slots = 0
    if school:
        sections = Section.objects.filter(is_active=True)
        for section in sections:
            periods = PeriodDefinition.objects.filter(school=school, wing=section.wing)
            periods_per_day = periods.count()
            teaching_periods = periods.filter(is_assembly=False, is_lunch=False, is_break=False)
            teaching_per_day = teaching_periods.count()
            if periods_per_day > 0:
                working_days = section.working_days or list(range(0, 6))
                total_slots += periods_per_day * len(working_days)
                teaching_slots += teaching_per_day * len(working_days)

    active_teacher_count = Teacher.objects.filter(is_active=True).count()
    average_per_teacher = round(teaching_slots / active_teacher_count, 1) if active_teacher_count > 0 else 0

    context = {
        'data': data,
        'total_required': total_required,
        'total_assigned': total_assigned,
        'total_remaining': total_required - total_assigned,
        'teaching_slots': teaching_slots,
        'total_slots': total_slots,
        'active_teacher_count': active_teacher_count,
        'average_per_teacher': average_per_teacher,
    }
    return render(request, 'timetable/free_slots.html', context)


def free_periods_by_date(request):
    school = School.objects.first()
    if not school:
        return render(request, 'timetable/free_by_date.html', {'days_with_indices': [], 'table': [], 'teachers': []})

    all_period_numbers = PeriodDefinition.objects.filter(school=school).values_list('period_number', flat=True).distinct() or range(1, 11)
    periods = [PeriodDefinition.objects.filter(school=school, period_number=pn).first() or {'period_number': pn, 'start_time': None, 'end_time': None} for pn in sorted(set(all_period_numbers))]

    days = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday']
    days_with_indices = [(i, name) for i, name in enumerate(days)]
    teachers = Teacher.objects.filter(is_active=True, school=school)

    date_str = request.GET.get('date')
    selected_weekday = None
    if date_str:
        try:
            selected_date = datetime.strptime(date_str, '%Y-%m-%d').date()
            selected_weekday = selected_date.weekday()
        except ValueError:
            pass

    if selected_weekday is not None:
        day_indices = [selected_weekday]
        day_names = [days[selected_weekday]]
    else:
        day_indices = [i for i, _ in days_with_indices]
        day_names = [name for _, name in days_with_indices]

    occupied = set(TimetableEntry.objects.all().values_list('teacher_id', 'day_of_week', 'period_number'))

    table = []
    for p in periods:
        pn = p.period_number if hasattr(p, 'period_number') else p['period_number']
        row = {'period': p, 'cells': []}
        for day_idx in day_indices:
            free_list = [t.name for t in teachers if (t.id, day_idx, pn) not in occupied]
            row['cells'].append(free_list)
        table.append(row)

    context = {
        'days_with_indices': list(zip(day_indices, day_names)),
        'table': table,
        'teachers': teachers,
        'selected_date': date_str,
    }
    return render(request, 'timetable/free_by_date.html', context)

def build_teacher_timetable_context(teacher):
    school = teacher.school or School.objects.first()
    if not school:
        return {'teacher': teacher, 'table': [], 'days_with_indices': []}

    periods = PeriodDefinition.objects.filter(school=school).order_by('period_number')
    if not periods:
        periods = [{'period_number': i, 'start_time': None, 'end_time': None} for i in range(1, 11)]

    days = ['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday']
    days_with_indices = [(i, name) for i, name in enumerate(days)]
    
    entries = TimetableEntry.objects.filter(teacher=teacher).select_related('class_instance', 'section', 'subject')

    grid = {}
    for e in entries:
        key = f"{e.day_of_week}_{e.period_number}"
        if key not in grid:
            grid[key] = []
        grid[key].append(e)

    processed_grid = {}
    for key, entry_list in grid.items():
        if len(entry_list) > 1 or entry_list[0].is_combined:
            first = entry_list[0]
            sections_str = ", ".join([f"{e.class_instance.name}-{e.section.name}" for e in entry_list])
            
            processed_grid[key] = {
                'id': first.id,
                'subject': first.subject,
                'teacher': first.teacher,
                'is_combined': True,
                'combined_display': f"{first.subject.name} ({sections_str})",
                'class_instance': first.class_instance,
                'section': first.section,
                'raw_entries': entry_list
            }
        else:
            processed_grid[key] = entry_list[0]

    table = []
    for p in periods:
        pn = p.period_number if hasattr(p, 'period_number') else p['period_number']
        row = {
            'period': p, 
            'cells': [processed_grid.get(f"{day_idx}_{pn}") for day_idx, _ in days_with_indices]
        }
        table.append(row)

    return {
        'teacher': teacher, 
        'table': table, 
        'days_with_indices': days_with_indices
    }

def export_pdf(request):

        return HttpResponse("No school found.", status=404)
    classes = Class.objects.filter(school=school, is_active=True).order_by('display_order')

    rows = [['Class', 'Section', 'Period'] + days]
    for cls in classes:
        for section in Section.objects.filter(class_instance=cls, is_active=True).order_by('name'):

                    row.append(f"{entry.subject.code}\n{entry.teacher.name}" if entry else "")
                rows.append(row)
